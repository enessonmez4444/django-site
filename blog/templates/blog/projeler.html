1. JSON Tabanlı Not Defteri

import json

class NotDefteri:
    def __init__(self):
        self.notlar = []
        self.dosya = "notlar.json"
        self.yukle()

    def not_ekle(self, baslik, icerik):
        yeni_not = {"baslik": baslik, "icerik": icerik}
        self.notlar.append(yeni_not)
        print("Not eklendi.")

    def notlari_goster(self):
        if len(self.notlar) == 0:
            print("Hiç not yok.")
            return
        for i, notum in enumerate(self.notlar, start=1):
            print(f"\n{i}. {notum['baslik']}")
            print(notum['icerik'])

    def not_sil(self, index):
        if index < 1 or index > len(self.notlar):
            print("Geçersiz numara")
            return
        silinen = self.notlar.pop(index - 1)
        print(f"{silinen['baslik']} silindi.")

    def kaydet(self):
        with open(self.dosya, "w", encoding="utf-8") as f:
            json.dump(self.notlar, f, ensure_ascii=False, indent=4)
        print("Notlar kaydedildi.")

    def yukle(self):
        try:
            with open(self.dosya, "r", encoding="utf-8") as f:
                self.notlar = json.load(f)
        except:
            self.notlar = []

def menu():
    uygulama = NotDefteri()
    while True:
        print("\n--- NOT DEFTERİ ---")
        print("1 - Not Ekle\n2 - Notları Göster\n3 - Not Sil\n4 - Kaydet\n5 - Çıkış")
        secim = input("Seçim: ")
        if secim == "1":
            uygulama.not_ekle(input("Başlık: "), input("İçerik: "))
        elif secim == "2":
            uygulama.notlari_goster()
        elif secim == "3":
            uygulama.notlari_goster()
            no = int(input("Silinecek no: "))
            uygulama.not_sil(no)
        elif secim == "4":
            uygulama.kaydet()
        elif secim == "5":
            uygulama.kaydet()
            break

menu()

class ATM:
    def __init__(self, isim, pin, bakiye):
        self.isim = isim
        self.pin = pin
        self.bakiye = bakiye

    def giris(self):
        deneme = 3
        while deneme > 0:
            girilen_pin = input("PIN giriniz: ")
            if girilen_pin == self.pin:
                print(f"Hoşgeldiniz {self.isim}")
                return True
            deneme -= 1
            print(f"Hatalı PIN. Kalan deneme: {deneme}")
        return False

    def bakiye_goster(self):
        print(f"Bakiyeniz: {self.bakiye} TL")

    def para_yatir(self, miktar):
        self.bakiye += miktar
        print("Para yatırıldı.")

    def para_cek(self, miktar):
        if miktar > self.bakiye:
            print("Yetersiz bakiye.")
        else:
            self.bakiye -= miktar
            print("Para çekildi.")

def menu():
    hesap = ATM("Enes", "1234", 1000)
    if hesap.giris():
        while True:
            print("\n--- ATM ---\n1-Bakiye 2-Yatır 3-Çek 4-Çıkış")
            s = input("Seçim: ")
            if s == "1": hesap.bakiye_goster()
            elif s == "2": hesap.para_yatir(int(input("Yatırılacak miktar: ")))
            elif s == "3": hesap.para_cek(int(input("Çekilecek miktar: ")))
            elif s == "4": break

menu()

class Hesap:
    def __init__(self, isim, sifre, bakiye=0):
        self.isim = isim
        self.sifre = sifre
        self.bakiye = bakiye

    def bakiye_goster(self):
        print(f"Bakiyeniz: {self.bakiye} TL")

    def para_yatir(self, miktar):
        self.bakiye += miktar
        print("Para yatırıldı.")

    def para_cek(self, miktar):
        if miktar > self.bakiye: print("Yetersiz bakiye")
        else:
            self.bakiye -= miktar
            print("Para çekildi.")

    def para_gonder(self, diger_hesap, miktar):
        if miktar > self.bakiye: print("Yetersiz bakiye")
        else:
            self.bakiye -= miktar
            diger_hesap.bakiye += miktar
            print("Para gönderildi.")

class Banka:
    def __init__(self):
        self.hesaplar = []

    def hesap_olustur(self):
        i, s = input("İsim: "), input("Şifre: ")
        self.hesaplar.append(Hesap(i, s))
        print("Hesap oluşturuldu.")

    def giris(self):
        i, s = input("İsim: "), input("Şifre: ")
        for h in self.hesaplar:
            if h.isim == i and h.sifre == s: return h
        return None

def menu():
    banka = Banka()
    while True:
        print("\n--- BANKA SİSTEMİ ---\n1-Hesap Aç 2-Giriş 3-Çıkış")
        sec = input("Seçim: ")
        if sec == "1": banka.hesap_olustur()
        elif sec == "2":
            h = banka.giris()
            if h:
                while True:
                    print("\n--- HESAP MENÜ ---\n1-Bakiye 2-Yatır 3-Çek 4-Gönder 5-Çıkış")
                    s2 = input("Seçim: ")
                    if s2 == "1": h.bakiye_goster()
                    elif s2 == "2": h.para_yatir(int(input("Miktar: ")))
                    elif s2 == "3": h.para_cek(int(input("Miktar: ")))
                    elif s2 == "4":
                        hedef = input("Alıcı: ")
                        m = int(input("Miktar: "))
                        for alici in banka.hesaplar:
                            if alici.isim == hedef:
                                h.para_gonder(alici, m)
                                break
                        else: print("Alıcı bulunamadı")
                    elif s2 == "5": break
        elif sec == "3": break

menu()

class Ogrenci:
    def __init__(self, isim, numara):
        self.isim = isim
        self.numara = numara
        self.notlar = []

    def not_ekle(self, notu):
        self.notlar.append(notu)
        print("Not eklendi")

    def ortalama_hesapla(self):
        if len(self.notlar) == 0: return 0
        return sum(self.notlar) / len(self.notlar)

    def bilgileri_goster(self):
        print(f"\nİsim: {self.isim}, No: {self.numara}, Ort: {self.ortalama_hesapla()}")

def menu():
    ogrenciler = []
    while True:
        print("\n--- ÖĞRENCİ OTOMASYONU ---\n1-Ekle 2-Not Gir 3-Listele 4-Çıkış")

        if sec == "1":
            ogrenciler.append(Ogrenci(input("İsim: "), input("No: ")))

5. XOR Şifrelemeli Not Sistemi

[KODU GÖR] 
import os

def sifrele(metin, key):
    # Metni basit XOR ile şifreler/çözer
    sonuc = ""
    for i in range(len(metin)):
        sonuc += chr(ord(metin[i]) ^ key)
    return sonuc

def kaydet(dosya_adi, metin, key):
    with open(dosya_adi, "w", encoding="utf-8") as f:
        f.write(sifrele(metin, key))
    print(f"Notunuz '{dosya_adi}' dosyasına kaydedildi.")

def oku(dosya_adi, key):
    if not os.path.exists(dosya_adi):
        print("Dosya bulunamadı!")
        return
    with open(dosya_adi, "r", encoding="utf-8") as f:
        icerik = f.read()
    print("Dosya içeriği:\n", sifrele(icerik, key))

def main():
    print("--- Şifreli Not Defteri ---")
    key = int(input("Şifreleme anahtarı (1-255): "))
    
    while True:
        print("\n1-Yaz/Kaydet 2-Oku 3-Çıkış")
        secim = input("Seçim: ")
        if secim == "1":
            dosya = input("Dosya adı: ")
            metin = input("Notunuz: ")
            kaydet(dosya, metin, key)
        elif secim == "2":
            dosya = input("Dosya adı: ")
            oku(dosya, key)
        elif secim == "3": break

if __name__ == "__main__":
    main()

6. Sezar Şifrelemeli Görev Listesi

import os

def sifrele(metin, key=3):
    return "".join([chr(ord(c)+key) for c in metin])

def cozumle(metin, key=3):
    return "".join([chr(ord(c)-key) for c in metin])

def ekle(gorev):
    with open("gorevler.txt","a", encoding="utf-8") as f:
        f.write(sifrele(gorev)+"\n")
    print("Görev eklendi.")

def listele():
    if not os.path.exists("gorevler.txt"):
        print("Görev yok.")
        return
    with open("gorevler.txt","r", encoding="utf-8") as f:
        for idx, satir in enumerate(f.readlines(),1):
            print(f"{idx}. {cozumle(satir.strip())}")

while True:
    secim = input("\n1-Ekle 2-Listele 3-Çıkış: ")
    if secim=="1":
        ekle(input("Görev: "))
    elif secim=="2":
        listele()
    elif secim=="3":
        break

import json, os

    






def harf_notu(not_ort):
    if not_ort>=90: return "AA"
    elif not_ort>=80: return "BA"
    elif not_ort>=70: return "BB"
    elif not_ort>=60: return "CB"
    elif not_ort>=50: return "CC"
    return "FF"

def kaydet(veri):
    with open("ogrenciler.json","w") as f:
        json.dump(veri,f)

def yukle():
    if not os.path.exists("ogrenciler.json"):
        return {}
    with open("ogrenciler.json","r") as f:
        return json.load(f)

ogrenciler = yukle()

while True:
